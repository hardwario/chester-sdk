110static inline int ctr_led_set(enum ctr_led_channel channel, bool is_on)
111{
112 switch (channel) {
113 case CTR_LED_CHANNEL_R:
114 channel = 0;
115 break;
116 case CTR_LED_CHANNEL_G:
117 channel = 1;
118 break;
119 case CTR_LED_CHANNEL_Y:
120 channel = 2;
121 break;
122 case CTR_LED_CHANNEL_EXT:
123 channel = 3;
124 break;
125 case CTR_LED_CHANNEL_LOAD:
126 channel = 4;
127 break;
128 }
129
130 if (is_on) {
131 return led_on(DEVICE_DT_GET(DT_NODELABEL(gpio_leds)), channel);
132 } else {
133 return led_off(DEVICE_DT_GET(DT_NODELABEL(gpio_leds)), channel);
134 }
135}
