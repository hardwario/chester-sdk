351/* %NCELLMEAS=5,7 (GCI search). rsrp and rsrq are the indexes the modem reports: RSRP in dBm is
352 * RSRP_IDX_TO_DBM of nrf (modem_info.h), RSRQ in dB is RSRQ_IDX_TO_DB of nrf, as the shell and
353 * ctr_lte_v2_conn_param (%CONEVAL) convert them */
354#define CTR_LTE_V2_NCELLMEAS_CELL_MAX 7
355#define CTR_LTE_V2_NCELLMEAS_NCELL_MAX 10

357struct ctr_lte_v2_ncellmeas_ncell {
358 uint32_t earfcn;
359 uint16_t pci;
360 int16_t rsrp; /* index, see above */
361 int16_t rsrq; /* index, see above */
362 int time_diff; /* ms between the measurement of its cell and this one, 0 if not valid */
363};
364
365struct ctr_lte_v2_ncellmeas_cell {
366 uint32_t eci;
367 uint32_t earfcn;
368 char plmn[7];
369 uint16_t tac;
370 uint16_t adv; /* timing advance as reported, 65535 if not valid */
371 uint16_t pci;
372 int16_t rsrp; /* index, see above */
373 int16_t rsrq; /* index, see above */
374 bool serving;
375 uint8_t neighbor_count; /* stored neighbours, at ncells[ncell_first] */
376 uint8_t ncell_first;
377};

379struct ctr_lte_v2_ncellmeas_param {
